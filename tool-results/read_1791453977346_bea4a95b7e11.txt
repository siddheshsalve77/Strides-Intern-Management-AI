    52→export function TasksView() {
    53→  const { currentEmployee, selectedTaskId, setSelectedTaskId, bumpDataVersion, dataVersion } = useApp()
    54→  const [tasks, setTasks] = useState<TaskT[]>([])
    55→  const [reportees, setReportees] = useState<EmployeeT[]>([])
    56→  const [loading, setLoading] = useState(true)
    57→  const [createOpen, setCreateOpen] = useState(false)
    58→
    59→  const load = useCallback(async () => {
    60→    if (!currentEmployee) return
    61→    // Server derives user from token; default returns manager's created tasks or employee's assigned tasks
    62→    const url = `/api/tasks`
    63→    const t = await authedFetch(url).then((r) => r.json()).catch(() => [])
    64→    setTasks(Array.isArray(t) ? t : [])
    65→    const isManager = ['MANAGER', 'HR', 'ADMIN'].includes(currentEmployee.role)
    66→    if (isManager) {
    67→      const r = await authedFetch(`/api/employees?managerId=me`).then((r) => r.json()).catch(() => [])
    68→      setReportees(Array.isArray(r) ? r : [])
    69→    }
    70→    setLoading(false)
    71→  }, [currentEmployee])
    72→
    73→  useEffect(() => { load() }, [currentEmployee?.id, dataVersion])
    74→
    75→  async function handleStatusChange(taskId: string, newStatus: string) {
    76→    setTasks((prev) => prev.map((t) => t.id === taskId ? { ...t, status: newStatus } : t))
    77→    const actionMap: Record<string, string> = {
    78→      CREATED: 'START', ASSIGNED: 'START', IN_PROGRESS: 'START',
    79→      SUBMITTED: 'SUBMIT', UNDER_REVIEW: 'SUBMIT',
    80→      APPROVED: 'APPROVE', COMPLETED: 'COMPLETE',
    81→    }
    82→    try {
    83→      const action = actionMap[newStatus] || 'START'
    84→      const res = await authedFetch(`/api/tasks/${taskId}`, {
    85→        method: 'PATCH', headers: { 'Content-Type': 'application/json' },
    86→        body: JSON.stringify({ action }),
    87→      })
    88→      if (!res.ok) throw new Error()
    89→      toast.success(`Task moved to ${newStatus.replace(/_/g, ' ').toLowerCase()}`)
    90→    } catch {
    91→      toast.error('Failed to update status')
    92→      load()
    93→    }
    94→  }
    95→
    96→  if (selectedTaskId) return <TaskDetail taskId={selectedTaskId} onBack={() => setSelectedTaskId(null)} onChange={load} />
    97→
    98→  return (
    99→    <div className="space-y-6">
   100→      <PageHeader
   101→        title="My Tasks"
   102→        description={currentEmployee && ['MANAGER', 'HR', 'ADMIN'].includes(currentEmployee.role) ? "Manage your team's work & assignments" : 'Track your assignments, deadlines and submissions'}
   103→        actions={
   104→          <Dialog open={createOpen} onOpenChange={setCreateOpen}>
   105→            <DialogTrigger asChild><Button className="gap-2"><Plus className="h-4 w-4" /> New Task</Button></DialogTrigger>
   106→            <CreateTaskDialog reportees={reportees} onClose={() => setCreateOpen(false)} onCreated={() => { load(); bumpDataVersion() }} />
   107→          </Dialog>
   108→        }
   109→      />
   110→
   111→      <div className="grid grid-cols-2 lg:grid-cols-4 gap-4">
   112→        <StatCard label="Total Tasks" value={tasks.length} icon={ListChecks} accent="primary" />
   113→        <StatCard label="In Progress" value={tasks.filter((t) => t.status === 'IN_PROGRESS').length} icon={Clock} accent="sky" />
   114→        <StatCard label="Completed" value={tasks.filter((t) => t.status === 'COMPLETED' || t.status === 'APPROVED').length} icon={CheckCircle2} accent="emerald" />
   115→        <StatCard label="Overdue" value={tasks.filter((t) => t.deadline && (daysUntil(t.deadline) ?? 1) < 0 && !['COMPLETED', 'APPROVED'].includes(t.status)).length} icon={AlertCircle} accent="rose" />
   116→      </div>
   117→
   118→      <TaskCompletionTrend />
   119→
   120→      <Tabs defaultValue="board">
   121→        <TabsList>
   122→          <TabsTrigger value="board">Board</TabsTrigger>
   123→          <TabsTrigger value="list">List</TabsTrigger>
   124→        </TabsList>
   125→
   126→        <TabsContent value="board">
   127→          {loading ? (
   128→            <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-4 gap-4">
   129→              {Array.from({ length: 8 }).map((_, i) => <div key={i} className="h-48 bg-muted animate-pulse rounded-lg" />)}
   130→            </div>
   131→          ) : tasks.length === 0 ? (
   132→            <Card><EmptyState icon={ListChecks} title="No tasks yet" description="Tasks assigned to you will appear here." /></Card>
   133→          ) : (
   134→            <KanbanBoard tasks={tasks} onTaskClick={(id) => setSelectedTaskId(id)} onStatusChange={handleStatusChange} />
   135→          )}
   136→        </TabsContent>
   137→
   138→        <TabsContent value="list">
   139→          {tasks.length === 0 ? <Card><EmptyState icon={ListChecks} title="No tasks" /></Card> : (
   140→            <div className="space-y-2">
   141→              {tasks.map((t) => <TaskCard key={t.id} task={t} onClick={() => setSelectedTaskId(t.id)} list />)}
   142→            </div>
   143→          )}
   144→        </TabsContent>
   145→      </Tabs>
   146→    </div>
   147→  )
   148→}

   150→function TaskCard({ task, onClick, list }: { task: TaskT; onClick?: () => void; list?: boolean }) {
   151→  const d = task.deadline ? daysUntil(task.deadline) : null
   152→  const overdue = d !== null && d < 0 && !['COMPLETED', 'APPROVED'].includes(task.status)
   153→  return (
   154→    <button onClick={onClick} className={`w-full text-left rounded-lg border border-border bg-card p-3 hover:shadow-md hover:border-primary/30 transition-all group ${list ? 'flex items-center gap-3' : ''}`}>
   155→      <div className={list ? 'flex-1 min-w-0' : ''}>
   156→        <div className="flex items-start justify-between gap-2 mb-1.5">
   157→          <p className="text-sm font-medium leading-snug group-hover:text-primary transition-colors">{task.title}</p>
   158→          <Badge variant="outline" className={`text-[10px] shrink-0 ${priorityColor(task.priority)}`}>{task.priority}</Badge>
   159→        </div>
   160→        {task.description && !list && <p className="text-xs text-muted-foreground line-clamp-2 mb-2">{task.description}</p>}
   161→        {task.project && <Badge variant="secondary" className="text-[10px] mb-2">{task.project.name}</Badge>}
   162→        <div className="flex items-center justify-between gap-2 mt-2">
   163→          <div className="flex items-center gap-2 min-w-0">
   164→            {task.assignee && (
   165→              <div className="flex items-center gap-1.5 min-w-0">
   166→                <Avatar className="h-5 w-5"><AvatarImage src={task.assignee.photoUrl} /><AvatarFallback className="text-[8px]">{initials(task.assignee.name)}</AvatarFallback></Avatar>
   167→                <span className="text-[11px] text-muted-foreground truncate">{task.assignee.name.split(' ')[0]}</span>
   168→              </div>
   169→            )}
   170→          </div>
   171→          <div className="flex items-center gap-1.5 shrink-0">
   172→            {task.deadline && (
   173→              <Badge variant="outline" className={`text-[10px] ${overdue ? 'bg-rose-500/15 text-rose-600 border-rose-500/20' : (d !== null && d <= 3) ? 'bg-amber-500/15 text-amber-600 border-amber-500/20' : ''}`}>
   174→                <Calendar className="h-2.5 w-2.5 mr-1" />{overdue ? 'Overdue' : d === 0 ? 'Today' : d !== null ? `${d}d` : ''}
   175→              </Badge>
   176→            )}
   177→            <Badge variant="outline" className={`text-[10px] ${taskColor(task.status)}`}>{taskLabel(task.status)}</Badge>
   178→          </div>
   179→        </div>
   180→      </div>
   181→    </button>
   182→  )
   183→}

   188→function CreateTaskDialog({ reportees, onClose, onCreated }: { reportees: EmployeeT[]; onClose: () => void; onCreated: () => void }) {
   189→  const { currentEmployee } = useApp()
   190→  const [title, setTitle] = useState('')
   191→  const [description, setDescription] = useState('')
   192→  const [assigneeId, setAssigneeId] = useState('') // Firebase UID
   193→  const [priority, setPriority] = useState('MEDIUM')
   194→  const [deadline, setDeadline] = useState('')
   195→  const [expectedDeliverable, setExpectedDeliverable] = useState('')
   196→  const [evaluationCriteria, setEvaluationCriteria] = useState('')
   197→  const [requirements, setRequirements] = useState<SubmissionRequirement[]>([
   198→    { id: `req_${Math.random().toString(36).slice(2, 10)}`, type: 'LONG_TEXT', label: 'Explanation', helpText: 'Describe your approach and any notes for the reviewer', required: true },
   199→  ])
   200→  const [submitting, setSubmitting] = useState(false)
   201→
   202→  function addRequirement() {
   203→    setRequirements([...requirements, {
   204→      id: `req_${Math.random().toString(36).slice(2, 10)}`,
   205→      type: 'TEXT', label: '', helpText: '', required: false,
   206→    }])
   207→  }
   208→  function updateReq(id: string, patch: Partial<SubmissionRequirement>) {
   209→    setRequirements(requirements.map(r => r.id === id ? { ...r, ...patch } : r))
   210→  }
   211→  function removeReq(id: string) {
   212→    setRequirements(requirements.filter(r => r.id !== id))
   213→  }
   214→
   215→  async function create() {
   216→    if (!currentEmployee || !title) { toast.error('Title required'); return }
   217→    if (!assigneeId) { toast.error('Please select an assignee'); return }
   218→    setSubmitting(true)
   219→    try {
   220→      // Filter out requirements without labels
   221→      const cleanReqs = requirements.filter(r => r.label.trim()).map(r => ({
   222→        ...r,
   223→        label: r.label.trim(),
   224→        helpText: (r.helpText || '').trim(),
   225→      }))
   226→      const res = await authedFetch('/api/tasks', {
   227→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   228→        body: JSON.stringify({
   229→          title, description,
   230→          assigneeId, // Firebase UID — server also verifies
   231→          priority,
   232→          deadline: deadline || undefined,
   233→          expectedDeliverable,
   234→          evaluationCriteria,
   235→          submissionRequirements: cleanReqs,
   236→        }),
   237→      })
   238→      if (!res.ok) {
   239→        const e = await res.json().catch(() => ({}))
   240→        throw new Error(e.error || 'Failed')
   241→      }
   242→      toast.success('Task created & assigned — assignee will be notified')
   243→      onClose(); onCreated()
   244→    } catch (e: any) { toast.error(e.message || 'Failed') } finally { setSubmitting(false) }
   245→  }
   246→
   247→  return (
   248→    <DialogContent className="max-w-2xl max-h-[92vh] overflow-y-auto">
   249→      <DialogHeader><DialogTitle>Create New Task</DialogTitle></DialogHeader>
   250→      <div className="space-y-4 py-2">
   251→        <div className="space-y-1.5">
   252→          <Label>Task Title</Label>
   253→          <Input value={title} onChange={(e) => setTitle(e.target.value)} placeholder="e.g. Build login API" />
   254→        </div>
   255→        <div className="space-y-1.5">
   256→          <Label>Description</Label>
   257→          <Textarea value={description} onChange={(e) => setDescription(e.target.value)} rows={3} placeholder="Detailed requirements, scope, acceptance criteria…" />
   258→        </div>
   259→        <div className="grid grid-cols-2 gap-3">
   260→          <div className="space-y-1.5">
   261→            <Label>Assign To</Label>
   262→            <Select value={assigneeId} onValueChange={setAssigneeId}>
   263→              <SelectTrigger><SelectValue placeholder="Select member" /></SelectTrigger>
   264→              <SelectContent>
   265→                {reportees.map((r) => <SelectItem key={r.uid || r.id} value={r.uid || r.id}>{r.name} · {r.designation}</SelectItem>)}
   266→              </SelectContent>
   267→            </Select>
   268→          </div>
   269→          <div className="space-y-1.5">
   270→            <Label>Priority</Label>
   271→            <Select value={priority} onValueChange={setPriority}>
   272→              <SelectTrigger><SelectValue /></SelectTrigger>
   273→              <SelectContent>
   274→                <SelectItem value="LOW">Low</SelectItem>
   275→                <SelectItem value="MEDIUM">Medium</SelectItem>
   276→                <SelectItem value="HIGH">High</SelectItem>
   277→                <SelectItem value="CRITICAL">Critical</SelectItem>
   278→              </SelectContent>
   279→            </Select>
   280→          </div>
   281→        </div>
   282→        <div className="grid grid-cols-2 gap-3">
   283→          <div className="space-y-1.5">
   284→            <Label>Deadline</Label>
   285→            <Input type="date" value={deadline} onChange={(e) => setDeadline(e.target.value)} />
   286→          </div>
   287→          <div className="space-y-1.5">
   288→            <Label>Expected Deliverable</Label>
   289→            <Input value={expectedDeliverable} onChange={(e) => setExpectedDeliverable(e.target.value)} placeholder="What should be delivered" />
   290→          </div>
   291→        </div>
   292→        <div className="space-y-1.5">
   293→          <Label>Evaluation Criteria (optional)</Label>
   294→          <Textarea value={evaluationCriteria} onChange={(e) => setEvaluationCriteria(e.target.value)} rows={2} placeholder="How will the submission be evaluated?" />
   295→        </div>
   296→
   297→        {/* Submission Requirements Builder */}
   298→        <div className="space-y-2 rounded-lg border border-border bg-muted/30 p-3">
   299→          <div className="flex items-center justify-between">
   300→            <div>
   301→              <p className="text-sm font-semibold flex items-center gap-2"><FileText className="h-4 w-4 text-primary" /> Submission Requirements</p>
   302→              <p className="text-xs text-muted-foreground">Define what the assignee must turn in. Like a Google Form.</p>
   303→            </div>
   304→            <Button type="button" variant="outline" size="sm" onClick={addRequirement} className="gap-1.5">
   305→              <Plus className="h-3.5 w-3.5" /> Add Field
   306→            </Button>
   307→          </div>
   308→          <div className="space-y-2">
   309→            {requirements.map((r, idx) => {
   310→              const typeCfg = REQ_TYPES.find(t => t.value === r.type)!
   311→              const Icon = typeCfg.icon
   312→              return (
   313→                <div key={r.id} className="rounded-lg border border-border bg-card p-3 space-y-2">
   314→                  <div className="flex items-center justify-between gap-2">
   315→                    <span className="text-xs text-muted-foreground">Field #{idx + 1}</span>
   316→                    <button type="button" onClick={() => removeReq(r.id)} className="text-muted-foreground hover:text-rose-600 transition-colors">
   317→                      <Trash2 className="h-3.5 w-3.5" />
   318→                    </button>
   319→                  </div>
   320→                  <div className="grid grid-cols-12 gap-2">
   321→                    <div className="col-span-5">
   322→                      <Label className="text-[10px]">Label</Label>
   323→                      <Input value={r.label} onChange={(e) => updateReq(r.id, { label: e.target.value })} placeholder="e.g. GitHub Repo URL" className="text-sm h-8" />
   324→                    </div>
   325→                    <div className="col-span-5">
   326→                      <Label className="text-[10px]">Type</Label>
   327→                      <Select value={r.type} onValueChange={(v) => updateReq(r.id, { type: v as SubmissionRequirementType })}>
   328→                        <SelectTrigger className="h-8 text-sm"><SelectValue /></SelectTrigger>
   329→                        <SelectContent>
   330→                          {REQ_TYPES.map((t) => (
   331→                            <SelectItem key={t.value} value={t.value}>
   332→                              <div className="flex items-center gap-2">
   333→                                <t.icon className="h-3.5 w-3.5" />
   334→                                <span>{t.label}</span>
   335→                              </div>
   336→                            </SelectItem>
   337→                          ))}
   338→                        </SelectContent>
   339→                      </Select>
   340→                    </div>
   341→                    <div className="col-span-2 flex items-end gap-2 pb-0.5">
   342→                      <label className="flex items-center gap-1.5 text-xs cursor-pointer">
   343→                        <input
   344→                          type="checkbox"
   345→                          checked={r.required}
   346→                          onChange={(e) => updateReq(r.id, { required: e.target.checked })}
   347→                          className="rounded border-border"
   348→                        />
   349→                        <span>Required</span>
   350→                      </label>
   351→                    </div>
   352→                  </div>
   353→                  <Input
   354→                    value={r.helpText || ''}
   355→                    onChange={(e) => updateReq(r.id, { helpText: e.target.value })}
   356→                    placeholder={`Help text — ${typeCfg.hint}`}
   357→                    className="text-xs h-8"
   358→                  />
   359→                  {['IMAGE', 'FILE', 'MULTIPLE_FILES'].includes(r.type) && (
   360→                    <div className="grid grid-cols-2 gap-2">
   361→                      <div>
   362→                        <Label className="text-[10px]">Max files</Label>
   363→                        <Input
   364→                          type="number"
   365→                          value={r.maxFiles || (r.type === 'MULTIPLE_FILES' ? 5 : 1)}
   366→                          onChange={(e) => updateReq(r.id, { maxFiles: parseInt(e.target.value) || 1 })}
   367→                          className="text-sm h-8"
   368→                          min={1}
   369→                          max={20}
   370→                        />
   371→                      </div>
   372→                      <div>
   373→                        <Label className="text-[10px]">Accepted types (comma-sep)</Label>
   374→                        <Input
   375→                          value={(r.acceptedTypes || []).join(',')}
   376→                          onChange={(e) => updateReq(r.id, { acceptedTypes: e.target.value.split(',').map(s => s.trim()).filter(Boolean) })}
   377→                          placeholder="image/png, application/pdf"
   378→                          className="text-sm h-8"
   379→                        />
   380→                      </div>
   381→                    </div>
   382→                  )}
   383→                  {['TEXT', 'LONG_TEXT', 'CODE'].includes(r.type) && (
   384→                    <div>
   385→                      <Label className="text-[10px]">Max length (chars, optional)</Label>
   386→                      <Input
   387→                        type="number"
   388→                        value={r.maxLength || ''}
   389→                        onChange={(e) => updateReq(r.id, { maxLength: parseInt(e.target.value) || undefined })}
   390→                        placeholder="e.g. 5000"
   391→                        className="text-sm h-8"
   392→                      />
   393→                    </div>
   394→                  )}
   395→                </div>
   396→              )
   397→            })}
   398→          </div>
   399→        </div>
   400→      </div>
   401→      <DialogFooter>
   402→        <Button variant="outline" onClick={onClose}>Cancel</Button>
   403→        <Button onClick={create} disabled={submitting} className="gap-2">
   404→          {submitting ? <Loader2 className="h-4 w-4 animate-spin" /> : <Send className="h-4 w-4" />}
   405→          Create & Assign
   406→        </Button>
   407→      </DialogFooter>
   408→    </DialogContent>
   409→  )
   410→}

   415→function TaskDetail({ taskId, onBack, onChange }: { taskId: string; onBack: () => void; onChange: () => void }) {
   416→  const { currentEmployee, bumpDataVersion } = useApp()
   417→  const [task, setTask] = useState<TaskT | null>(null)
   418→  const [loading, setLoading] = useState(true)
   419→  const [submissionNote, setSubmissionNote] = useState('')
   420→  const [managerScore, setManagerScore] = useState(80)
   421→  const [managerFeedback, setManagerFeedback] = useState('')
   422→  const [aiLoading, setAiLoading] = useState(false)
   423→  const [celebrate, setCelebrate] = useState(0)
   424→  const [celebrateMsg, setCelebrateMsg] = useState('')
   425→  const [acting, setActing] = useState(false)

   591→  // Build the submission payload (used by both saveDraft and submitFinal)
   592→  function buildSubmissionPayload(status: 'DRAFT' | 'SUBMITTED') {
   593→    const fields = Object.entries(fieldValues).map(([reqId, fv]) => ({
   594→      requirementId: reqId,
   595→      value: fv.value,
   596→      files: fv.files,
   597→    }))
   598→    // Flatten files + links for backward compat
   599→    const allFiles: SubmissionFile[] = []
   600→    const allLinks: { url: string; label?: string; requirementId?: string }[] = []
   601→    if (task?.submissionRequirements) {
   602→      for (const req of task.submissionRequirements) {
   603→        const fv = fieldValues[req.id]
   604→        if (!fv) continue
   605→        if (['IMAGE', 'FILE', 'MULTIPLE_FILES'].includes(req.type)) {
   606→          for (const f of fv.files) allFiles.push({ ...f, requirementId: req.id })
   607→        }
   608→        if (['URL', 'GITHUB_URL'].includes(req.type) && fv.value) {
   609→          allLinks.push({ url: fv.value, label: req.label, requirementId: req.id })
   610→        }
   611→      }
   612→    }
   613→    return { taskId, submissionNote, fields, files: allFiles, links: allLinks, status }
   614→  }

   616→  // Validate required fields before submission
   617→  function validateSubmission(): { ok: boolean; missing: SubmissionRequirement[] } {
   618→    if (!task?.submissionRequirements) return { ok: true, missing: [] }
   619→    const missing: SubmissionRequirement[] = []
   620→    for (const req of task.submissionRequirements) {
   621→      if (!req.required) continue
   622→      const fv = fieldValues[req.id]
   623→      if (!fv) { missing.push(req); continue }
   624→      if (['TEXT', 'LONG_TEXT', 'CODE', 'URL', 'GITHUB_URL'].includes(req.type)) {
   625→        if (!fv.value.trim()) missing.push(req)
   626→      }
   627→      if (['IMAGE', 'FILE', 'MULTIPLE_FILES'].includes(req.type)) {
   628→        if (fv.files.length === 0) missing.push(req)
   629→      }
   630→    }
   631→    return { ok: missing.length === 0, missing }
   632→  }

  1119→function SubmissionContent({ submission, onDownloadFile, compact }: { submission: any; onDownloadFile: (path: string) => void; compact?: boolean }) {
  1120→  if (!submission) return null
  1121→  const fields: any[] = Array.isArray(submission.fields) ? submission.fields : []
  1122→  const files: any[] = Array.isArray(submission.files) ? submission.files : []
  1123→  const links: any[] = Array.isArray(submission.links) ? submission.links : []
  1124→
  1125→  // Legacy: files/links as strings (comma-separated)
  1126→  const legacyFiles = typeof submission.files === 'string' ? submission.files.split(',').filter(Boolean) : []
  1127→  const legacyLinks = typeof submission.links === 'string' ? submission.links.split(',').filter(Boolean) : []
  1128→
  1129→  if (fields.length === 0 && legacyFiles.length === 0 && legacyLinks.length === 0 && !submission.submissionNote) return null
  1130→
  1131→  return (
  1132→    <div className={`mt-2 space-y-2 ${compact ? '' : 'rounded-lg border border-border p-3'}`}>
  1133→      {fields.map((f: any, i: number) => (
