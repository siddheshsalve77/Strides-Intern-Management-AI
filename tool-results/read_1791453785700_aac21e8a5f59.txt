   384→                  <span className="text-[12px] text-muted-foreground">{l.value}/{l.total}</span>

   387→                  <div className={`h-full ${l.color} rounded-full transition-all`} style={{ width: `${(l.value / l.total) * 100}%` }} />
   388→                </div>
   389→              </div>
   390→            ))}
   391→          </div>
   392→        </div>

   395→      {/* ============================================================
   396→          FOURTH ROW — Goals + Holidays + Announcements

   398→      <div className="grid grid-cols-1 lg:grid-cols-3 gap-4">

   410→              </div>
   411→              <Progress value={g.progress} className="h-1" />
   412→            </div>
   413→          )) : <p className="text-[12px] text-muted-foreground py-4">No active goals</p>}

   415→
   416→        {/* Upcoming Holidays */}
   417→        <div>

   419→            <p className="text-[14px] font-semibold">Holidays</p>

   423→            {upcomingHolidays.slice(0, 3).map((h) => (

   433→              </div>
   434→            ))}
   435→            {upcomingHolidays.length === 0 && <p className="text-[12px] text-muted-foreground py-4">No upcoming holidays</p>}
   436→          </div>
   437→        </div>
   438→
   439→        {/* Announcements */}
   440→        <div>
   441→          <div className="flex items-center justify-between mb-3">
   442→            <p className="text-[14px] font-semibold">Announcements</p>
   443→            <button onClick={() => setView('announcements')} className="text-[11px] text-primary hover:underline">All</button>
   444→          </div>
   445→          <div className="space-y-2">
   446→            {announcements.slice(0, 3).map((a) => (
   447→              <div key={a.id} onClick={() => setView('announcements')} className="cursor-pointer rounded-md px-2 py-1.5 hover:bg-accent/40 transition-colors">

   449→                <p className="text-[10px] text-muted-foreground truncate">{a.authorName} · {fmtRelative(a.publishedAt)}</p>
   450→              </div>
   451→            ))}
   452→            {announcements.length === 0 && <p className="text-[12px] text-muted-foreground py-4">No announcements</p>}
   453→          </div>
   454→        </div>
   455→      </div>
   456→
   457→      {/* Activity Feed — full width, no card wrapper */}
   458→      <ActivityFeed />
   459→    </div>
   460→  )
   461→}

   464→function ActivityFeed() {
   465→  const { currentEmployee } = useApp()
   466→  const [activities, setActivities] = useState<any[]>([])
   467→
   468→  useEffect(() => {
   469→    if (!currentEmployee) return
   470→    const isManager = ['MANAGER', 'HR', 'ADMIN'].includes(currentEmployee.role)
   471→    const url = isManager ? `/api/activity?scope=org&managerId=${currentEmployee.id}&limit=8` : `/api/activity?limit=8`
   472→    authedFetch(url).then(r => r.ok ? r.json() : []).then(a => setActivities(Array.isArray(a) ? a : [])).catch(() => {})
